cmake_minimum_required(VERSION 3.16)
project(why-slow VERSION 1.0.0 LANGUAGES CXX)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
  message(FATAL_ERROR "why-slow supports Linux only")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
include(GNUInstallDirs)
option(WHY_SLOW_SANITIZE "Enable address and undefined behavior sanitizers" OFF)
option(WHY_SLOW_LEAK "Enable standalone leak sanitizer" OFF)
if(WHY_SLOW_SANITIZE AND WHY_SLOW_LEAK)
  message(FATAL_ERROR "Select one sanitizer configuration")
endif()
add_compile_options(-Wall -Wextra -Wpedantic -Werror)
if(WHY_SLOW_SANITIZE)
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
  add_link_options(-fsanitize=address,undefined)
endif()
if(WHY_SLOW_LEAK)
  add_compile_options(-fsanitize=leak -fno-omit-frame-pointer)
  add_link_options(-fsanitize=leak)
endif()
add_library(why_slow_core src/parsers.cpp src/sampling.cpp src/source.cpp src/context.cpp src/analysis.cpp src/format.cpp)
target_include_directories(why_slow_core PUBLIC include)
add_executable(why-slow src/main.cpp)
target_link_libraries(why-slow PRIVATE why_slow_core)
install(TARGETS why-slow RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(FILES man/why-slow.1 DESTINATION ${CMAKE_INSTALL_MANDIR}/man1)
install(FILES README.md LICENSE DESTINATION ${CMAKE_INSTALL_DOCDIR})
install(DIRECTORY docs/ DESTINATION ${CMAKE_INSTALL_DOCDIR})
include(CTest)
if(BUILD_TESTING)
  add_executable(unit_tests tests/unit.cpp)
  target_link_libraries(unit_tests PRIVATE why_slow_core)
  add_test(NAME unit COMMAND unit_tests)
  set_tests_properties(unit PROPERTIES TIMEOUT 30)
  find_package(Python3 COMPONENTS Interpreter REQUIRED)
  add_test(NAME integration COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/integration.py $<TARGET_FILE:why-slow>)
  set_tests_properties(integration PROPERTIES TIMEOUT 60)
endif()
set(CPACK_PACKAGE_CONTACT "guhshosting")
set(CPACK_PACKAGE_VENDOR "guhshosting")
set(CPACK_RPM_PACKAGE_DESCRIPTION "Collect and explain local Linux performance observations without network access, telemetry, configuration changes, automatic optimization, or unsupported causal claims.")
set(CPACK_PACKAGE_DESCRIPTION_SUMMARY "Read-only Linux performance observations with cautious explanations")
set(CPACK_PACKAGE_VERSION ${PROJECT_VERSION})
set(CPACK_PACKAGE_FILE_NAME "why-slow-${PROJECT_VERSION}-${CMAKE_SYSTEM_PROCESSOR}")
set(CPACK_DEBIAN_PACKAGE_MAINTAINER "guhshosting")
set(CPACK_DEBIAN_PACKAGE_SECTION "utils")
set(CPACK_DEBIAN_PACKAGE_SHLIBDEPS ON)
set(CPACK_RPM_PACKAGE_LICENSE "MIT")
set(CPACK_RPM_PACKAGE_GROUP "Applications/System")
set(CPACK_RPM_PACKAGE_AUTOREQ ON)
include(CPack)
